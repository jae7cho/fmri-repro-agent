"""Agent prototype v2 (2026-11-09): an agent reconstructs a paper's preprocessing with tools; code
checks its evidence. An experiment outside the pipeline: it imports the repo's text loader, methods
finder, citation lookup, span resolver and the defaults KB, and changes none of them.

v2 against v1 (v1 spent 42 calls on one paper, 34 of its 85 tool calls keyword searches):
- the first message carries the text the pipeline's extraction reads (``find_methods_section`` cut
  at the reference list, plus the supplement's MRI section, as ``batch.process_paper`` builds it),
  capped at EXCERPT_MAX characters; tools are for what that text does not settle;
- every field is given in the one ``submit`` call (v1: one ``record_field`` call per field); a
  submit missing scored fields is answered once with their names, and the next submit is final.
- references / open_reference take a doc: an opened paper's own reference list (its GROBID TEI, or
  the JATS ref-list of Europe PMC full text) can be searched and opened, up to MAX_HOP_DEPTH
  citations from the paper; a reference with no DOI gets the one CrossRef records for it, as the
  pipeline's hop does (crossref_doi).
Kept from v1: the tools, the provenances, the checks, 60 turns, a forced submit at $2.50, the last
KEEP_TOOL_RESULTS tool results in full.

Scored fields use the repo's names: motion_correction (performed, method, tool, reference_scan,
interpolation, similarity_metric, susceptibility_correction), spatial_smoothing (performed, fwhm_mm,
kernel_type, space), temporal_filtering (performed, pass_type, band_or_cutoff, filter_type, order);
every step carries applied_by. Six provenances: EXTRACTED, CITED, DEFERRED_UNREAD (the paper defers
and the source could not be read; the quote is the paper's deferral sentence), KB_DEFAULT, MODEL,
NOT_REPORTED. The prompt also tells the agent to look up a named pipeline's realignment in the KB.
Other steps are free-form.

Checks (code, after the run):
- EXTRACTED / CITED: the quote resolves in the named document with ``span_resolver.resolve_quote``
  (its citation-marker tier lets a quote skip reference markers); a number in the value must be a
  whole token of the quote.
- KB_DEFAULT: some kb_lookup result in the run's trace names the value.

    python agent_proto_v2.py --paper X.pdf --paper-id x --out x.json
"""

def _fn(name: str, desc: str, props: dict[str, Any], req: list[str]) -> dict[str, Any]:
    return {"type": "function", "function": {"name": name, "description": desc,
            "parameters": {"type": "object", "properties": props, "required": req}}}

MAX_HOP_DEPTH = 2  # the paper -> a cited paper -> a paper that one cites


def jats_bibliography(xml: bytes) -> dict[str, BiblEntry]:
    """A JATS article's <ref-list> as BiblEntry rows (ref id, first surnames, year, title, DOI)."""
    import xml.etree.ElementTree as ET

    out: dict[str, BiblEntry] = {}
    try:
        root = ET.fromstring(xml)
    except ET.ParseError:
        return out
    for i, ref in enumerate(root.iter("ref")):
        rid = ref.get("id") or f"r{i}"

        def txt(path: str) -> str | None:
            el = ref.find(path)
            return " ".join("".join(el.itertext()).split()) if el is not None else None

        surnames = tuple(" ".join("".join(s.itertext()).split()) for s in ref.iter("surname"))
        doi = next((" ".join("".join(p.itertext()).split()) for p in ref.iter("pub-id")
                    if p.get("pub-id-type") == "doi"), None)
        out[rid] = BiblEntry(rid, txt(".//article-title"), surnames, txt(".//year"), doi)
    return out


class Library:
    def __init__(self, pdf: Path) -> None:
        text, parser, _ = load_text(pdf)
        text, _ = reflow_one_word_lines(text)
        supp_rec, supp = load_supplement(pdf)
        # the excerpt, as batch.process_paper builds the extraction text
        methods = find_methods_section(text, text_source=parser)
        body_end = _body_end(text)
        excerpt = methods.text
        if methods.start_offset + len(methods.text) > body_end:
            excerpt = methods.text[: max(0, body_end - methods.start_offset)]
        marker = supplement_marker(Path(supp_rec["path"]).name)
        if supp:
            supp_body = supp[: _body_end(supp)]
            excerpt += marker + find_supplement_mri_section(supp_body).text
        self.methods_found_via = methods.found_via
        self.excerpt_chars = len(excerpt)
        self.excerpt = excerpt[:EXCERPT_MAX]
        if supp:
            text += marker + supp
        self.docs: dict[str, str] = {"paper": text}
        self.titles: dict[str, str] = {"paper": pdf.name}
        self.bibs: dict[str, dict[str, Any]] = {"paper": bibliography(fetch_tei(pdf)[0])}
        self.depth: dict[str, int] = {"paper": 0}
        self.pdf = pdf
        self.opened: list[dict[str, Any]] = []
        self.kb_results: list[str] = []

    def _doc(self, d: str) -> str:
        if d not in self.docs:
            raise KeyError(f"unknown doc {d!r}; open docs: {list(self.docs)}")
        return self.docs[d]

    def outline(self, doc: str) -> Any:
        out, off = [], 0
        for line in self._doc(doc).split("\n"):
            s = line.strip()
            if 0 < len(s) < 90 and not s.endswith("."):
                out.append({"offset": off, "heading": s})
            off += len(line) + 1
        return out[:150]

    def search(self, doc: str, query: str) -> Any:
        t = self._doc(doc)
        words = [w.lower() for w in query.split()]
        hits = []
        for m in re.finditer(r"[^.\n]*(?:\.(?!\s)[^.\n]*)*[.\n]?", t):
            s = m.group(0)
            if s.strip() and all(w in s.lower() for w in words):
                hits.append({"offset": m.start(), "sentence": s.strip()[:400]})
                if len(hits) >= 12:
                    break
        return hits or "no hits"

    def read(self, doc: str, start: int, length: int = 4000) -> Any:
        s = max(0, int(start))
        return self._doc(doc)[s : s + min(int(length or 4000), 4000)]

    def _bib(self, doc: str) -> dict[str, Any]:
        self._doc(doc)
        return self.bibs.get(doc) or {}

    def references(self, query: str, doc: str = "paper") -> Any:
        """The reference list of any open doc (the paper's, or a cited paper's own)."""
        bib = self._bib(doc)
        if not bib:
            return f"no reference list read for {doc!r}"
        rows = []
        for b in bib.values():
            n = re.sub(r"\D", "", b.bibl_id)
            rows.append({"ref_id": b.bibl_id, "number_if_numbered": int(n) + 1 if n and b.bibl_id.startswith("b") else None,
                         "authors": ", ".join(b.authors[:3]), "year": b.year, "title": b.title})
        words = query.lower().split()
        if not words:
            return rows[:40]
        hits = [r for r in rows if all(w in json.dumps(r).lower() for w in words)]  # every word, any order
        return hits[:20] or "no match"

    def open_reference(self, ref_id: str, doc: str = "paper") -> Any:
        """Open a reference of any open doc, up to MAX_HOP_DEPTH citations from the paper."""
        b = self._bib(doc).get(ref_id)
        if b is None:
            return f"no reference {ref_id!r} in {doc!r}"
        depth = self.depth[doc] + 1
        if depth > MAX_HOP_DEPTH:
            return f"cannot open: {doc!r} is already {self.depth[doc]} citations from the paper (limit {MAX_HOP_DEPTH})"
        doc_id = f"ref:{ref_id}" if doc == "paper" else f"{doc}/{ref_id}"
        if doc_id not in self.docs:
            state, path, confirmed = resolve_local_pdf(_bibl_ref_entry(b, "paper"), CORPUS_DIRS)
            if path is not None:
                text, _, _ = load_text(path)
                text, _ = reflow_one_word_lines(text)
                src = f"local_pdf:{path.name}:{confirmed}"
                try:
                    bib = bibliography(fetch_tei(path)[0])
                except Exception:
                    bib = {}
            else:
                # as the pipeline's hop: the entry's own DOI, else the one CrossRef records for it
                doi, doi_src = b.doi, "entry"
                if not doi:
                    try:
                        doi, doi_src = crossref_doi(_bibl_ref_entry(b, "paper")), "crossref"
                    except Exception:
                        doi = None
                if not doi:
                    self.opened.append({"doc": doc_id, "depth": depth, "source": f"not opened ({state}; no DOI from entry or CrossRef)"})
                    return f"cannot open: local lookup {state}; no DOI in the entry or from CrossRef"
                got = europepmc_fulltext(doi, self.pdf.parent)
                if got is None:
                    self.opened.append({"doc": doc_id, "depth": depth, "source": f"not opened ({state}; DOI from {doi_src}; no Europe PMC text)"})
                    return f"cannot open: local lookup {state}; Europe PMC has no full text for its DOI"
                text, src = got[0], f"europepmc (DOI from {doi_src})"
                bib = jats_bibliography(got[1].read_bytes())
            self.docs[doc_id] = text
            self.bibs[doc_id] = bib
            self.depth[doc_id] = depth
            self.titles[doc_id] = b.title or ref_id
            self.opened.append({"doc": doc_id, "depth": depth, "source": src, "n_references": len(bib)})
        return {"doc": doc_id, "title": self.titles[doc_id], "chars": len(self.docs[doc_id]),
                "references": len(self.bibs.get(doc_id) or {})}

    def kb_lookup(self, kind: str, name: str, version: str | None = None, fields: str | None = None) -> Any:
        try:
            if kind == "pipeline":
                out: Any = kb.recognize_with_provenance(name)
            elif kind == "dataset":
                out = _plain(kb.recognize_dataset(name, version))
            elif kind == "package":
                out = kb.recognize_tool(name)
            elif kind == "program":
                out = kb.recognize_program(name)
            elif kind == "realignment":
                out = (_plain(kb.realignment_tool_at(name, version)) if version else
                       {"program": _plain(kb.realignment_program(name)), "defaults": kb.realignment_defaults(name)})
            elif kind == "params":
                fs = [f.strip() for f in (fields or "").split(",") if f.strip()]
                out = _plain(kb.get_param_defaults(name, version or "", fs))
            else:
                out = f"unknown kind {kind!r}"
        except Exception as exc:  # the KB's errors are answers
            out = f"{type(exc).__name__}: {exc}"
        self.kb_results.append(json.dumps(out, default=str))
        return out


def _plain(o: Any) -> Any:
    if o is None or isinstance(o, (str, int, float, bool)):
        return o
    if isinstance(o, dict):
        return {k: _plain(v) for k, v in o.items()}
    if isinstance(o, (list, tuple, set, frozenset)):
        return [_plain(v) for v in o]
    if hasattr(o, "model_dump"):
        return o.model_dump()
    if hasattr(o, "__dataclass_fields__"):
        return {k: _plain(getattr(o, k)) for k in o.__dataclass_fields__}
    return str(o)


_NUMS = re.compile(r"\d+(?:\.\d+)?")


def check_field(f: dict[str, Any], lib: Library) -> str:
    prov = f.get("provenance")
    if prov in ("EXTRACTED", "CITED", "DEFERRED_UNREAD"):
        doc = f.get("doc") or ("paper" if prov in ("EXTRACTED", "DEFERRED_UNREAD") else "")
        q = (f.get("quote") or "").strip()
        if not q:
            return "no_quote"
        if prov == "DEFERRED_UNREAD":
            if doc != "paper":
                return "deferral_not_in_paper_doc"
            res = resolve_quote(q, lib.docs["paper"])
            return "quote_not_found" if res.failure_reason == "quote_not_found" else "deferral_quote_resolved"
        if prov == "EXTRACTED" and doc != "paper":
            return "extracted_from_non_paper_doc"
        if prov == "CITED" and not doc.startswith("ref:"):
            return "cited_without_reference_doc"
        if doc not in lib.docs:
            return "unknown_doc"
        res = resolve_quote(q, lib.docs[doc])
        if res.failure_reason == "quote_not_found":
            return "quote_not_found"
        qn = {float(x) for x in _NUMS.findall(q)}
        if any(float(x) not in qn for x in _NUMS.findall(str(f.get("value") or ""))):
            return "number_not_in_quote"
        return "quote_ambiguous" if res.failure_reason == "quote_ambiguous" else (
            "quote_resolved_recovered" if res.recovered else "quote_resolved")
    if prov == "KB_DEFAULT":
        v = str(f.get("value") or "").lower()
        toks = [t for t in re.findall(r"[a-z0-9]+", v) if len(t) > 1]
        hit = any(toks and all(t in r.lower() for t in toks) for r in lib.kb_results)
        return "kb_result_names_value" if hit else "kb_result_does_not_name_value"
    return "n/a"


KEEP_TOOL_RESULTS = 8
_ELIDED = "[elided: an earlier tool result; call the tool again if you need it]"


def compact(msgs: list[dict[str, Any]]) -> None:
    """Keep the last KEEP_TOOL_RESULTS tool results in full; older ones become a stub (the
    provider's per-minute token quota throttled whole-history turns, 2026-11-09). The first message
    (the methods text) and the agent's own calls are kept."""
    tool_idx = [i for i, m in enumerate(msgs) if m.get("role") == "tool"]
    for i in tool_idx[:-KEEP_TOOL_RESULTS]:
        if msgs[i]["content"] != _ELIDED and len(msgs[i]["content"]) > 200:
            msgs[i]["content"] = _ELIDED


def run(pdf: Path, paper_id: str) -> dict[str, Any]:
    lib = Library(pdf)
    fields: dict[str, dict[str, Any]] = {}
    msgs: list[dict[str, Any]] = [
        {"role": "system", "content": SYSTEM},
        {"role": "user", "content": (
            f"Paper {paper_id}. The whole paper is open as doc 'paper' ({len(lib.docs['paper'])} characters). "
            f"Its methods text ({lib.methods_found_via}; {lib.excerpt_chars} characters"
            f"{', first ' + str(EXCERPT_MAX) + ' shown' if lib.excerpt_chars > EXCERPT_MAX else ''}) follows. "
            "Reconstruct its preprocessing.\n\n<methods>\n" + lib.excerpt + "\n</methods>")},
    ]
    cost, calls, trace, submitted, forced, order = 0.0, 0, [], False, None, None
    throttled, error, asked_missing, disconnected = 0, None, None, 0
    t0 = time.time()
    turn = 0
    for turn in range(MAX_TURNS):
        last = turn == MAX_TURNS - 1
        if last or cost >= FORCE_AT_USD:
            forced = forced or ("turn_cap" if last else "cost_cap")
            choice: Any = {"type": "function", "function": {"name": "submit"}}
        else:
            choice = "auto"
        compact(msgs)
        resp = None
        for wait in (0, 15, 30, 60, 90, 120, 180, 240, 300, 300, 300):  # about 27 minutes in all
            time.sleep(wait)
            try:
                resp = completion(model=MODEL, messages=msgs, tools=TOOLS, tool_choice=choice,
                                  temperature=0.0, max_tokens=8000)
                break
            except RateLimitError:
                throttled += 1
                print(f"  {paper_id} throttled", flush=True)
            except APIConnectionError:
                # 2026-11-10: a network outage ended the batch (DNS failures); wait it out the same way
                disconnected += 1
                print(f"  {paper_id} connection error", flush=True)
        if resp is None:
            error = "rate_limited_or_disconnected"
            break
        calls += 1
        try:
            cost += float(completion_cost(completion_response=resp))
        except Exception:
            pass
        m = resp.choices[0].message
        msgs.append(m.model_dump(exclude_none=True))
        tcs = m.tool_calls or []
        if not tcs:
            msgs.append({"role": "user", "content": "Use the tools; call submit when done."})
            continue
        for c in tcs:
            try:
                args = json.loads(c.function.arguments or "{}")
            except json.JSONDecodeError:
                args = {}
            name = c.function.name
            trace.append({"turn": turn, "tool": name,
                          "args": {k: v for k, v in args.items() if k != "fields"},
                          **({"n_fields": len(args.get("fields") or [])} if name == "submit" else {})})
            if name == "submit":
                for f in args.get("fields") or []:
                    if isinstance(f, dict):
                        key = f"{f.get('step')}.{f.get('field')}"
                        fields[key] = {k: f.get(k) for k in ("step", "field", "value", "provenance", "doc", "quote", "note")}
                order = args.get("order") or order
                missing = [f"{s}.{k}" for s, ks in SCHEMA.items() for k in ks if f"{s}.{k}" not in fields]
                if missing and asked_missing is None and not forced:
                    asked_missing = missing
                    out: Any = ("Not final: these scored fields are missing (give each, NOT_REPORTED if "
                                f"nothing says it): {', '.join(missing)}. Call submit again; fields you "
                                "already gave are kept.")
                else:
                    submitted = True
                    out = "submitted"
            else:
                try:
                    out = getattr(lib, name)(**args)
                except Exception as exc:
                    out = f"error: {type(exc).__name__}: {exc}"
            msgs.append({"role": "tool", "tool_call_id": c.id, "content": json.dumps(out, default=str)[:12000]})
        print(f"  {paper_id} turn {turn}: {[c.function.name for c in tcs]} ${cost:.4f}", flush=True)
        if submitted:
            break
    for f in fields.values():
        f["check"] = check_field(f, lib)
    return {
        "paper_id": paper_id, "model": MODEL, "cost_usd": round(cost, 4), "calls": calls,
        "turns": turn + 1, "wall_s": round(time.time() - t0, 1), "submitted": submitted,
